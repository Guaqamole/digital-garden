1000명 거르는 query
```sql
select shop_id, count(*) as cnt  
from namkyu.fs_order_goods_for_recommend  
group by shop_id  
having count(*) > 2  
order by count(*) desc  
limit 1000
```



```sql
select  
    sb.biz_cate3_code as biz_cate_code,  
    '' as vendor_id,  
    og.goods_category2_id as goods_category_id,  
    og.goods_id,  
    rank() over(PARTITION BY biz_cate3_code, goods_category2_id ORDER BY sum(qty) desc) as ranking,  
    sum(og.qty) as qty  
    #og.qty / sum(og.qty) over(partition by biz_cate3_code, goods_category2_id) as score  
from fs_order_goods_for_recommend og  
join (select * from fs_shop_biz_category where biz_cate3_code is not null) sb on sb.shop_id = og.shop_id  
group by biz_cate3_code, goods_category_id, goods_id  
order by biz_cate_code, goods_category_id, qty desc
;
```

## Command
```python
# airflow
use tmux
airflow scheduler -D
airflow webserver -D
airflow celery flower
airflow celery worker # at node1
airflow celery worker # at node 2

# kafka
use tmux
sudo ./bin/kafka-server-start.sh ./config/kraft/server.properties # at node 1
sudo ./bin/kafka-server-start.sh ./config/kraft/server.properties # at node 2

# hadoop (namenode: linux)
$HADOOP_HOME/sbin/start-dfs.sh
$HADOOP_HOME/sbin/start-yarn.sh

$HADOOP_HOME/sbin/stop-all.sh

# spark (master: linux)
$SPARK_HOME/sbin/start-all.sh
$SPARK_HOME/sbin/start-history-server.sh

$SPARK_HOME/sbin/stop-all.sh
$SPARK_HOME/sbin/stop-history-server.sh

# hbase
tmux
$HBASE_HOME/bin/start-hbase.sh
$HBASE_HOME/bin/stop-hbase.sh

# hive
$HIVE_HOME/bin/hive

# flink
$FLINK_HOME/bin/start-cluster.sh
$FLINK_HOME/bin/stop-cluster.sh
```

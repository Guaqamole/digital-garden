# Iris Classification
![](https://i.imgur.com/B8IXiqJ.png)


```python
from sklearn.datasets import load_iris  
import pandas as pd 

iris = load_iris()
 

# 데이터셋을 DataFrame으로 변환합니다.
iris_df = pd.DataFrame(iris.data, columns=iris.feature_names)
iris_df['target'] = iris.target
display(iris_df.head())

from sklearn.ensemble import RandomForestClassifier   

# 독립변수, 종속변수 얻기 
X, y = iris.data, iris.target 

# 모델 선택 및 생성
model = RandomForestClassifier(random_state = 42) 

# 모델 학습
model.fit(X, y)

# 새로운 데이터에 대한 예측 
# sepal length = 5.1cm, sepal width = 3.5cm	, petal length = 1.4cm	,petal width = 0.2c인 새로운 
# 데이터가 입력 된경우

new_data = [[5.1, 3.5, 1.4, 0.2]]  # 새로운 붓꽃의 특성 값
prediction = model.predict(new_data)
display(f"prediction : {prediction}")
display("Predicted class:", iris.target_names[prediction])

'prediction : [0]'
'Predicted class:'
array(['setosa'], dtype='<U10')
```

## Precision, Recall, F1 핵심
- Accuracy: 클래스 불균형이 심할때 모델을 잘못 평가할수있기 때문에 단순히 정확도만 높다고해서 좋지않음. 
- Precision: 스팸을 못잡지만, 정상중요한 메일이 스팸 폴더로 이동하지 않게 하는것
- Recall: 모든 스팸을 잡지만, 정상 이메일을 스팸으로 잘못 플래그 하는경우
- f1-score: 재현, 정밀의 균형도

#### DecisionTreeClassifier 검증 (Wine Dataset)
```python
import pandas as pd
train_wine = pd.read_csv('train_wine.csv')

train_x = train_wine.drop('quality', axis = 1 )
train_y = train_wine['quality'] 

from sklearn.model_selection import train_test_split
from sklearn.tree import DecisionTreeClassifier
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score

# 학습 데이터와 검증 데이터로 분할
X_train, X_val, y_train, y_val = train_test_split(train_x, train_y, test_size=0.25, random_state=42)

# Decision Tree Classifier 모델 생성 및 학습
clf_wine_DT = DecisionTreeClassifier(random_state=42)
clf_wine_DT.fit(X_train, y_train)

# 검증 데이터에 대한 예측 수행
y_pred = clf_wine_DT.predict(X_val)

# 성능 지표 계산 및 출력
accuracy_wine_DT =  accuracy_score(y_pred, y_val)
precision_wine_DT =  precision_score(y_pred, y_val, average='macro')
recall_wine_DT =  recall_score(y_pred, y_val, average='macro')
f1_score_wine_DT =  f1_score(y_pred, y_val, average='macro')

display(f"Accuracy:{accuracy_wine_DT}")
display(f"precision:{precision_wine_DT}")
display(f"recall: {recall_wine_DT}")
display(f"f1_score: {f1_score_wine_DT}")
```
